cmake_minimum_required(VERSION 3.20)
project(LSHdrSession LANGUAGES CXX)
add_executable(LSHdrSession WIN32 main.cpp)
target_compile_features(LSHdrSession PRIVATE cxx_std_17)
target_compile_definitions(LSHdrSession PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(LSHdrSession PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/../../build/native")
if(LSC_MINHOOK_SOURCE AND EXISTS "${LSC_MINHOOK_SOURCE}/CMakeLists.txt")
  enable_language(C)
  add_subdirectory("${LSC_MINHOOK_SOURCE}" minhook EXCLUDE_FROM_ALL)
  add_executable(LSC_HDRProbeTest probe_test.cpp ../addon_manager/hdr_probe.cpp)
  target_compile_features(LSC_HDRProbeTest PRIVATE cxx_std_17)
  target_compile_definitions(LSC_HDRProbeTest PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_include_directories(LSC_HDRProbeTest PRIVATE ../addon_manager)
  target_link_libraries(LSC_HDRProbeTest PRIVATE d3d11 dxgi d3dcompiler minhook)
  set_target_properties(LSC_HDRProbeTest PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/../../build/native")
endif()
