cmake_minimum_required(VERSION 3.20)
project(LSC_AMDNR LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "AMD NR requires Windows x64")
endif()
if(NOT LSC_UPSTREAM_SOURCE)
  message(FATAL_ERROR "Pass the pinned addon manager source as LSC_UPSTREAM_SOURCE")
endif()
if(NOT LSC_AMD_RUNTIME_SOURCE)
  message(FATAL_ERROR "Pass the pinned, adapter-patched AMD runtime source as LSC_AMD_RUNTIME_SOURCE")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set(upscalers "${LSC_UPSTREAM_SOURCE}/addons/Upscalers/src")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/runtime.lock.json" runtime_pin)
string(JSON model_sha GET "${runtime_pin}" model sha256)
add_library(amd_pipeline STATIC src/pipeline.cpp "${upscalers}/engine/flow_estimator.cpp")
target_compile_definitions(amd_pipeline PRIVATE LSC_AMD_MODEL_SHA256="${model_sha}")
target_include_directories(amd_pipeline PUBLIC src vendor "${upscalers}" "${LSC_UPSTREAM_SOURCE}/manager/sdk/include")
target_link_libraries(amd_pipeline PUBLIC d3d11 d3d12 dxgi d3dcompiler bcrypt)
add_library(LSC_AMDNR SHARED src/addon.cpp)
target_link_libraries(LSC_AMDNR PRIVATE amd_pipeline)
set(bundle "${CMAKE_CURRENT_SOURCE_DIR}/../../build/native/LSC-AMDNR")
add_subdirectory("${LSC_AMD_RUNTIME_SOURCE}/runtime" "${CMAKE_BINARY_DIR}/amd_runtime" EXCLUDE_FROM_ALL)
target_sources(dlss5nr_runtime PRIVATE src/runtime_marker.cpp)
set_target_properties(dlss5nr_runtime PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${bundle}/runtime")
target_compile_options(dlss5nr_runtime PRIVATE /MP4)
add_dependencies(LSC_AMDNR dlss5nr_runtime)
set_target_properties(LSC_AMDNR PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${bundle}")
add_custom_command(TARGET LSC_AMDNR POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/addon.json" "$<TARGET_FILE_DIR:LSC_AMDNR>/addon.json")
add_executable(LSC_AMDNRProbe src/probe.cpp)
target_link_libraries(LSC_AMDNRProbe PRIVATE amd_pipeline)
add_library(lsc_probe_runtime SHARED src/probe_runtime.cpp)
target_include_directories(lsc_probe_runtime PRIVATE vendor)
set_target_properties(lsc_probe_runtime PROPERTIES OUTPUT_NAME "dlss5nr_runtime" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${bundle}/probe-fixture")
add_dependencies(LSC_AMDNRProbe lsc_probe_runtime)
add_executable(LSC_AMDAdapterTest src/adapter_test.cpp)
set_target_properties(LSC_AMDAdapterTest PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${bundle}")
set_target_properties(LSC_AMDNRProbe PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${bundle}")
if(MSVC)
  target_compile_options(amd_pipeline PRIVATE /W4 /EHsc /MP4)
  target_compile_options(LSC_AMDNR PRIVATE /W4 /EHsc)
endif()
